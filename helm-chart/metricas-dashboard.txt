# Métricas do KSM -> Boas para capacidade, requests e limits

# CPU Utilization (Requests/Allocatable) por Node
(
  sum by (node) (kube_pod_container_resource_requests{resource="cpu"}) 
  / 
  kube_node_status_allocatable{resource="cpu"}
) * 100

# Memory Utilization (Requests/Allocatable) por Node  
(
  sum by (node) (kube_pod_container_resource_requests{resource="memory"}) 
  / 
  kube_node_status_allocatable{resource="memory"}
) * 100

# Pods Running por Node
count by (node) (kube_pod_status_phase{phase="Running"})
